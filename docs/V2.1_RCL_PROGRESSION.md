#### 2. Link Controller Logic
**New file needed**: `simulation/link.controller.js`

```javascript
class LinkController {
    static run(room) {
        if (Game.time % 5 !== 0) return; // Throttle to every 5 ticks
        
        const links = room.find(FIND_MY_STRUCTURES, {
            filter: s => s.structureType === STRUCTURE_LINK
        });
        
        if (links.length < 2) return;
        
        // Find source link (has energy) and target link (near controller)
        const sourceLink = links.find(l => l.store[RESOURCE_ENERGY] > 400);
        const targetLink = links.find(l => l.store.getFreeCapacity(RESOURCE_ENERGY) > 400);
        
        if (sourceLink && targetLink && sourceLink.cooldown === 0) {
            sourceLink.transferEnergy(targetLink);
        }
    }
}
```

#### 1. Mineral Harvesting
**New role needed**: `simulation/role.miner.js`

```javascript
class RoleMiner {
    static run(creep) {
        const mineral = creep.room.find(FIND_MINERALS)[0];
        
        if (!mineral || mineral.mineralAmount === 0) {
            // No mineral, become upgrader
            creep.memory.role = 'upgrader';
            return;
        }
        
        if (creep.store.getFreeCapacity() === 0) {
            // Deliver to storage
            const storage = creep.room.storage;
            if (creep.transfer(storage, mineral.mineralType) === ERR_NOT_IN_RANGE) {
                creep.moveTo(storage);
            }
        } else {
            // Mine mineral
            if (creep.harvest(mineral) === ERR_NOT_IN_RANGE) {
                creep.moveTo(mineral);
            }
        }
    }
}

// Optimal miner body:
```

```javascript
        // Select 2 input labs and 1+ output labs
```
